grant select, insert, update, delete on public.vinted_investments to authenticated; grant select, insert, update, delete on public.vinted_investment_categories to authenticated; grant select, insert, update, delete on public.vinted_items to authenticated; create or replace view public.vinted_investment_summary with (security_invoker = true) as select i.id, i.user_id, i.name, i.invested_amount, i.article_count, i.investment_date, i.note, i.created_at, i.updated_at, count(distinct v.id) as linked_articles, count(distinct v.id) filter (where v.status = 'sold') as sold_articles, coalesce( sum( greatest( coalesce(v.price, 0) - coalesce(v.fees, 0), 0 ) ) filter (where v.status = 'sold'), 0 ) as total_net_sales, least( i.invested_amount, coalesce( sum( greatest( coalesce(v.price, 0) - coalesce(v.fees, 0), 0 ) ) filter (where v.status = 'sold'), 0 ) ) as capital_recovered, greatest( i.invested_amount - coalesce( sum( greatest( coalesce(v.price, 0) - coalesce(v.fees, 0), 0 ) ) filter (where v.status = 'sold'), 0 ), 0 ) as remaining_capital, greatest( coalesce( sum( greatest( coalesce(v.price, 0) - coalesce(v.fees, 0), 0 ) ) filter (where v.status = 'sold'), 0 ) - i.invested_amount, 0 ) as profit, case when i.invested_amount > 0 then least( 100, ( coalesce( sum( greatest( coalesce(v.price, 0) - coalesce(v.fees, 0), 0 ) ) filter (where v.status = 'sold'), 0 ) / i.invested_amount ) * 100 ) else 0 end as recovery_percent, case when i.invested_amount > 0 then ( ( coalesce( sum( greatest( coalesce(v.price, 0) - coalesce(v.fees, 0), 0 ) ) filter (where v.status = 'sold'), 0 ) - i.invested_amount ) / i.invested_amount ) * 100 else 0 end as profitability_percent, case when i.invested_amount <= 0 then 'active' when coalesce( sum( greatest( coalesce(v.price, 0) - coalesce(v.fees, 0), 0 ) ) filter (where v.status = 'sold'), 0 ) >= i.invested_amount then 'recovered' else 'active' end as status from public.vinted_investments i left join public.vinted_items v on v.investment_id = i.id where i.user_id = auth.uid() group by i.id, i.user_id, i.name, i.invested_amount, i.article_count, i.investment_date, i.note, i.created_at, i.updated_at; create or replace view public.vinted_financial_dashboard with (security_invoker = true) as select coalesce(sum(invested_amount), 0) as total_invested, coalesce(sum(capital_recovered), 0) as total_recovered, coalesce(sum(remaining_capital), 0) as total_remaining, coalesce(sum(profit), 0) as total_profit, case when coalesce(sum(invested_amount), 0) > 0 then ( coalesce(sum(profit), 0) / sum(invested_amount) ) * 100 else 0 end as total_profitability_percent, count(*) filter (where status = 'active') as active_investments, count(*) filter (where status = 'recovered') as recovered_investments from public.vinted_investment_summary; grant select on public.vinted_investment_summary to authenticated; grant select on public.vinted_financial_dashboard to authenticated;
